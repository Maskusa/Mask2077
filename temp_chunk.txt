    }
    return null;
  })();
  if (!summaryText) {
    return null;
  }
  const match = summaryText.match(/listening port:\s*(\d+)/i);
  if (!match) {
    return null;
  }
  const parsed = Number.parseInt(match[1], 10);
  return Number.isFinite(parsed) ? parsed : null;
};
